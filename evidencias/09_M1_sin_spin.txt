import rclpy
from mi_primer_paquete.mi_nodo import MiNodo

rclpy.init()
nodo = MiNodo()
nodo.get_logger().info('Nodo creado. NO se llama a rclpy.spin()')
nodo.destroy_node()
rclpy.shutdown()
